pragma solidity ^0.8.24;

/// Minimal self-contained ERC-721 drop with public paid mint, ERC-2981 royalties
/// and a platform fee split. Deployed directly by the creator's wallet.
contract LilyApeDrop {
    string public name;
    string public symbol;
    string private _baseURI;
    address public owner;
    address public platformWallet;
    uint16 public platformFeeBps;
    uint256 public maxSupply;
    uint256 public mintPrice;
    uint256 public maxPerTx;
    uint256 public totalSupply;
    bool public mintOpen = true;
    uint96 private _royaltyBps;

    mapping(uint256 => address) private _owners;
    mapping(address => uint256) private _balances;
    mapping(uint256 => address) private _approvals;
    mapping(address => mapping(address => bool)) private _operators;

    event Transfer(address indexed from, address indexed to, uint256 indexed tokenId);
    event Approval(address indexed owner, address indexed approved, uint256 indexed tokenId);
    event ApprovalForAll(address indexed owner, address indexed operator, bool approved);

    modifier onlyOwner() { require(msg.sender == owner, "not owner"); _; }

    constructor(
        string memory n, string memory s, string memory baseURI_,
        uint256 maxSupply_, uint256 price_, uint256 maxPerTx_,
        uint96 royaltyBps_, address platform_, uint16 platformFeeBps_
    ) {
        require(royaltyBps_ <= 1000 && platformFeeBps_ <= 2000, "fee");
        name = n; symbol = s; _baseURI = baseURI_;
        maxSupply = maxSupply_; mintPrice = price_; maxPerTx = maxPerTx_;
        _royaltyBps = royaltyBps_; platformWallet = platform_; platformFeeBps = platformFeeBps_;
        owner = msg.sender;
    }

    function mint(uint256 qty) external payable {
        require(mintOpen, "closed");
        require(qty > 0 && qty <= maxPerTx, "qty");
        require(totalSupply + qty <= maxSupply, "sold out");
        require(msg.value == mintPrice * qty, "price");
        for (uint256 i = 0; i < qty; i++) {
            uint256 id = ++totalSupply;
            _owners[id] = msg.sender;
            emit Transfer(address(0), msg.sender, id);
        }
        _balances[msg.sender] += qty;
        if (msg.value > 0) {
            uint256 fee = (msg.value * platformFeeBps) / 10000;
            if (fee > 0) { (bool a,) = platformWallet.call{value: fee}(""); require(a, "fee tx"); }
            (bool b,) = owner.call{value: msg.value - fee}(""); require(b, "pay tx");
        }
    }

    function setMintOpen(bool v) external onlyOwner { mintOpen = v; }
    function setBaseURI(string calldata u) external onlyOwner { _baseURI = u; }
    function transferOwnership(address n) external onlyOwner { owner = n; }

    function tokenURI(uint256 id) external view returns (string memory) {
        require(_owners[id] != address(0), "none");
        return string(abi.encodePacked(_baseURI, _toString(id), ".json"));
    }
    function royaltyInfo(uint256, uint256 salePrice) external view returns (address, uint256) {
        return (owner, (salePrice * _royaltyBps) / 10000);
    }
    function supportsInterface(bytes4 i) external pure returns (bool) {
        return i == 0x01ffc9a7 || i == 0x80ac58cd || i == 0x5b5e139f || i == 0x2a55205a;
    }

    function balanceOf(address a) external view returns (uint256) { require(a != address(0)); return _balances[a]; }
    function ownerOf(uint256 id) public view returns (address o) { o = _owners[id]; require(o != address(0), "none"); }
    function approve(address to, uint256 id) external {
        address o = ownerOf(id);
        require(msg.sender == o || _operators[o][msg.sender], "auth");
        _approvals[id] = to; emit Approval(o, to, id);
    }
    function getApproved(uint256 id) external view returns (address) { ownerOf(id); return _approvals[id]; }
    function setApprovalForAll(address op, bool v) external { _operators[msg.sender][op] = v; emit ApprovalForAll(msg.sender, op, v); }
    function isApprovedForAll(address o, address op) external view returns (bool) { return _operators[o][op]; }

    function transferFrom(address from, address to, uint256 id) public {
        address o = ownerOf(id);
        require(o == from && to != address(0), "from");
        require(msg.sender == o || _approvals[id] == msg.sender || _operators[o][msg.sender], "auth");
        delete _approvals[id];
        _balances[from]--; _balances[to]++; _owners[id] = to;
        emit Transfer(from, to, id);
    }
    function safeTransferFrom(address from, address to, uint256 id) external { safeTransferFrom(from, to, id, ""); }
    function safeTransferFrom(address from, address to, uint256 id, bytes memory data) public {
        transferFrom(from, to, id);
        if (to.code.length > 0) {
            (bool ok, bytes memory r) = to.call(abi.encodeWithSelector(0x150b7a02, msg.sender, from, id, data));
            require(ok && r.length >= 32 && abi.decode(r, (bytes4)) == 0x150b7a02, "receiver");
        }
    }

    function _toString(uint256 v) internal pure returns (string memory) {
        if (v == 0) return "0";
        uint256 t = v; uint256 d;
        while (t != 0) { d++; t /= 10; }
        bytes memory b = new bytes(d);
        while (v != 0) { d--; b[d] = bytes1(uint8(48 + v % 10)); v /= 10; }
        return string(b);
    }
}
